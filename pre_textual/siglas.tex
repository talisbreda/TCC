% Lista de abreviaturas e siglas
\begin{siglas}
    \item[BFS] Busca em largura (\textit{Breadth-First Search})
    \item[DFS] Busca em profundidade (\textit{Depth-First Search})
    \item[FIFO] Primeiro a entrar, primeiro a sair (\textit{First In, First Out})
    \item[LAPJV] Algoritmo de Jonker-Volgenant para o LAP
    \item[LBAP] Problema de atribuição gargalo linear (\textit{Linear Bottleneck Assignment Problem})
    \item[LIFO] Último a entrar, primeiro a sair (\textit{Last In, First Out})
    \item[MCBM] Emparelhamento bipartido de cardinalidade máxima (\textit{Maximum Cardinality Bipartite Matching})
    \item[MCM] Emparelhamento de cardinalidade máxima (\textit{Maximum Cardinality Matching})
    \item[MIM] Emparelhamento induzido máximo (\textit{Maximum Induced Matching})
    \item[MIS] Conjunto independente máximo (\textit{Maximum Independent Set})
    \item[MVC] Cobertura mínima de vértices (\textit{Minimum Vertex Cover})
    \item[NP] Tempo polinomial não determinístico (\textit{Nondeterministic Polynomial time})
    \item[P] Tempo polinomial (\textit{Polynomial time})
    \item[PL] Programação linear
    \item[QAP] Problema de atribuição quadrática (\textit{Quadratic Assignment Problem})
    \item[SMP] Problema do casamento estável (\textit{Stable Marriage Problem})
    \item[SSP] Caminhos mínimos sucessivos (\textit{Successive Shortest Paths})
\end{siglas}
