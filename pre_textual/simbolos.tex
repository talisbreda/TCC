% Lista de símbolos
\begin{simbolos}
    \item[$G=(V,E)$] Grafo com conjunto de vértices $V$ e conjunto de arestas $E$
    \item[$D=(V,A)$] Grafo direcionado com conjunto de arcos $A$
    \item[$\{u,v\}$] Aresta não direcionada entre os vértices $u$ e $v$
    \item[$(u,v)$] Arco direcionado de $u$ para $v$
    \item[$V_1, V_2$] Classes de cor (partes) de um grafo bipartido
    \item[$w(e)$] Peso (ou custo) da aresta $e$
    \item[$M$] Emparelhamento
    \item[$|M|$] Cardinalidade do emparelhamento $M$
    \item[$v(G)$] Cardinalidade máxima de um emparelhamento em $G$
    \item[$M \ominus P$] Diferença simétrica entre os conjuntos de arestas $M$ e $P$
    \item[$\delta(u,v)$] Peso de um caminho mínimo de $u$ a $v$
    \item[$\delta(v)$] Conjunto das arestas incidentes ao vértice $v$
    \item[$E(U)$] Conjunto das arestas com ambos os extremos em $U \subseteq V$
    \item[$x_e$] Variável de decisão associada à aresta $e$ em um modelo de programação linear
    \item[$y_v, z_B$] Variáveis duais associadas ao vértice $v$ e ao conjunto ímpar $B$
    \item[$c(u,v)$] Capacidade do arco $(u,v)$ em uma rede de fluxo
    \item[$G_f$] Rede residual associada ao fluxo $f$
    \item[$c_f(u,v)$] Capacidade residual do arco $(u,v)$
    \item[$s, t$] Vértices fonte e sorvedouro de uma rede de fluxo
    \item[$L(G)$] Grafo de linha de $G$
    \item[$L(G)^2$] Quadrado do grafo de linha de $G$
    \item[$O(\cdot)$] Limite assintótico superior
    \item[$n$] Tamanho da instância
    \item[$k$] Expoente empírico de crescimento
    \item[$R^2$] Coeficiente de determinação do ajuste do expoente empírico
\end{simbolos}
